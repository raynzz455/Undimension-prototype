   407→// DYNAMIC LIST — reusable for work history, education, skills
   408→// ═══════════════════════════════════════════════════════════════════════════
   409→function DynamicList<T extends Record<string, unknown>>({
   410→  title, items, setItems, fields,
   411→}: {
   412→  title: string;
   413→  items: T[];
   414→  setItems: (items: T[]) => void;
   415→  fields: { key: string; placeholder: string; width?: string; textarea?: boolean; type?: string }[];
   416→}) {
   417→  const update = (i: number, key: string, v: string | number | boolean) => setItems(items.map((item, idx) => (idx === i ? { ...item, [key]: v } : item)));
   418→  const add = () => { const newItem = {} as T; fields.forEach((f) => { (newItem as Record<string, unknown>)[f.key] = f.type === "number" ? 0 : ""; }); setItems([...items, newItem]); };
   419→  const remove = (i: number) => setItems(items.filter((_, idx) => idx !== i));
   420→
   421→  return (
   422→    <div className="mb-3">
   423→      <div className="flex items-center justify-between mb-2">
   424→        <span className="font-mono-ud text-xs font-bold text-[#d4ff00] tracking-wider">{title} ({items.length})</span>
   425→        <button onClick={add} className="flex items-center gap-1 bg-[#d4ff00] text-black font-mono-ud text-xs font-black px-2 py-1 border-2 border-[#d4ff00] hover:bg-transparent hover:text-[#d4ff00] transition-colors no-color-transition min-h-[36px]"><Plus className="w-3 h-3" /> ADD</button>
   426→      </div>
   427→      <div className="space-y-2">
   428→        {items.map((item, i) => (
   429→          <div key={i} className="border-2 border-[#d4ff00]/30 p-2 bg-[#1a1a1a] relative">
   430→            <div className="grid grid-cols-1 md:grid-cols-6 gap-2">
   431→              {fields.map((f) => (
   432→                <div key={f.key} className={f.width || ""}>
   433→                  {f.textarea ? (
   434→                    <textarea value={String(item[f.key] ?? "")} onChange={(e) => update(i, f.key, e.target.value)} placeholder={f.placeholder} rows={1} className="w-full px-2 py-1.5 bg-black border border-[#d4ff00]/30 text-white font-mono-ud text-xs resize-none focus:outline-none focus:border-[#ff00ff]" />
   435→                  ) : (
   436→                    <input value={String(item[f.key] ?? "")} onChange={(e) => update(i, f.key, f.type === "number" ? Number(e.target.value) : e.target.value)} placeholder={f.placeholder} type={f.type || "text"} className="w-full px-2 py-1.5 bg-black border border-[#d4ff00]/30 text-white font-mono-ud text-xs focus:outline-none focus:border-[#ff00ff]" />
   437→                  )}
   438→                </div>
   439→              ))}
   440→            </div>
   441→            <button onClick={() => remove(i)} className="absolute top-1 right-1 p-1.5 bg-[#ff4d4d]/80 hover:bg-[#ff4d4d] text-white border border-white/30 transition-colors no-color-transition" title="Remove"><Trash2 className="w-3 h-3" /></button>
   442→          </div>
   443→        ))}
   444→      </div>
   445→    </div>
   446→  );
   447→}
