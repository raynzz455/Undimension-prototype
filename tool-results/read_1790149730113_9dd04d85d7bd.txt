   161→function AchievementImageCarousel({ images, color }: { images: string[]; color: string }) {
   162→  const [idx, setIdx] = useState(0);
   163→
   164→  const next = () => setIdx((p) => (p + 1) % images.length);

   173→      <div className="relative aspect-[4/3] overflow-hidden">
   174→        <img
   175→          src={images[idx]}
   176→          alt={`Achievement evidence ${idx + 1}`}
   177→          className="w-full h-full object-cover"
   178→          key={idx}
   179→        />

   191→              onClick={next}

   213→          ))}

   220→function AchievementModal({
   221→  achievement,
   222→  memberColor,
   223→  onClose,
   224→}: {
   225→  achievement: Achievement | null;
   226→  memberColor: string;
   227→  onClose: () => void;
   228→}) {
   229→  const open = achievement !== null;
   230→  const panelRef = useRef<HTMLDivElement>(null);
   231→  useFocusTrap(panelRef, open);
   232→
   233→  useEffect(() => {
   234→    if (!open) return;
   235→    const onKey = (e: KeyboardEvent) => {
   236→      if (e.key === "Escape") {
   237→        document.body.classList.remove("modal-open");
   238→        onClose();
   239→      }
   240→    };
   241→    window.addEventListener("keydown", onKey);
   242→    return () => window.removeEventListener("keydown", onKey);
   243→  }, [open, onClose]);
   244→
   245→  // Hide navbar when modal is open
   246→  useEffect(() => {
   247→    if (open) {
   248→      document.body.classList.add("modal-open");
   249→    } else {
   250→      document.body.classList.remove("modal-open");
   251→    }
   252→    return () => { document.body.classList.remove("modal-open"); };
   253→  }, [open]);
   254→
   255→  return (
   256→    <AnimatePresence>
   257→      {open && achievement && (
   258→        <motion.div
   259→          className="fixed inset-0 z-[85] flex items-center justify-center p-4 md:p-8"
   260→          initial={{ opacity: 0 }}
   261→          animate={{ opacity: 1 }}
   262→          exit={{ opacity: 0 }}
   263→          transition={{ duration: 0.2 }}
   264→        >
   265→          <div className="absolute inset-0 bg-black/80 backdrop-blur-sm" onClick={onClose} />
   266→
   267→          {/* Relative wrapper */}
   268→          <div className="relative w-full max-w-md">
   269→            {/* X Close */}
   270→            <button
   271→              onClick={onClose}
   272→              className="absolute -top-4 -right-4 z-30 w-10 h-10 flex items-center justify-center bg-[#ff4d4d] text-white border-4 border-black dark:border-white shadow-[4px_4px_0_#000] dark:shadow-[4px_4px_0_#fff] hover:rotate-90 transition-transform no-color-transition"
   273→              aria-label="Close"
   274→            >
   275→              <X className="w-5 h-5" />
   276→            </button>
   277→
   278→          <motion.div
   279→            ref={panelRef}
   280→            role="dialog"
   281→            aria-modal="true"
   282→            aria-labelledby="ud-achievement-title"
   283→            tabIndex={-1}
   284→            className="relative w-full max-h-[90vh] overflow-y-auto border-8 border-black dark:border-white bg-white dark:bg-[#09090b] shadow-[8px_8px_0_#000] md:shadow-[16px_16px_0_#000] dark:md:shadow-[16px_16px_0_#d4ff00] outline-none"
   285→            initial={{ scale: 0.85, y: 30, rotate: -2 }}
   286→            animate={{ scale: 1, y: 0, rotate: 0 }}
   287→            exit={{ scale: 0.85, y: 30, rotate: -2 }}
   288→            transition={{ type: "spring", stiffness: 300, damping: 24 }}
   289→            onAnimationComplete={() => {
   290→              if (panelRef.current) panelRef.current.scrollTop = 0;
   291→            }}
   292→          >
   293→
   294→            {/* Colored header banner */}
   295→            <div className="p-6 border-b-8 border-black dark:border-white" style={{ backgroundColor: memberColor }}>
   296→              <div className="flex items-center justify-center mb-2">
   297→                <div className="w-16 h-16 flex items-center justify-center bg-black border-4 border-black">
   298→                  <Trophy className="w-8 h-8 text-white" />
   299→                </div>
   300→              </div>
   301→              <div className="font-mono-ud text-sm font-black tracking-[0.3em] uppercase text-black/60 text-center">
   302→                ▸ ACHIEVEMENT UNLOCKED
   303→              </div>
   304→            </div>
   305→
   306→            {/* Body */}
   307→            <div className="p-6">
   308→              <h3
   309→                id="ud-achievement-title"
   310→                className="font-bebas text-4xl md:text-5xl text-black dark:text-white leading-none mb-2 text-center"
   311→              >
   312→                {achievement.title}
   313→              </h3>
   314→              <div className="font-mono-ud text-xs font-black text-black/50 dark:text-white/50 tracking-widest mb-4 text-center">
   315→                ◆ {achievement.year} ◆
   316→              </div>
   317→              <div className="border-4 border-black dark:border-white bg-black dark:bg-white p-4 mb-4">
   318→                <p className="font-mono-ud text-sm text-white dark:text-black leading-relaxed text-center">
   319→                  {achievement.description}
   320→                </p>
   321→              </div>
   322→              {/* Achievement photo carousel (sertifikat, medali, foto kemenangan) */}
