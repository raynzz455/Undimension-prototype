   160→function AchievementImageCarousel({ images, color }: { images: string[]; color: string }) {
   161→  const [idx, setIdx] = useState(0);
   162→
   163→  const next = () => setIdx((p) => (p + 1) % images.length);
   164→  const prev = () => setIdx((p) => (p - 1 + images.length) % images.length);
   165→
   166→  return (
   167→    <div className="mb-4 border-4 border-black dark:border-white bg-black overflow-hidden relative">
   168→      <div className="font-mono-ud text-sm font-black tracking-[0.2em] uppercase text-white/60 px-3 py-1 border-b-2 border-white/20 flex items-center justify-between">
   169→        <span>▸ EVIDENCE / CERTIFICATE</span>
   170→        <span>{idx + 1}/{images.length}</span>
   171→      </div>
   172→      <div className="relative aspect-[4/3] overflow-hidden">
   173→        <img
   174→          src={images[idx]}
   175→          alt={`Achievement evidence ${idx + 1}`}
   176→          className="w-full h-full object-cover"
   177→          key={idx}
   178→        />
   179→        <div className="absolute inset-0 ud-scanlines opacity-20 pointer-events-none" />
   180→        {images.length > 1 && (
   181→          <>
   182→            <button
   183→              onClick={prev}
   184→              className="absolute left-1 top-1/2 -translate-y-1/2 w-8 h-8 flex items-center justify-center bg-black/70 border-2 border-white text-white hover:bg-white hover:text-black transition-colors"
   185→              aria-label="Previous image"
   186→            >
   187→              <ChevronLeft className="w-4 h-4" />
   188→            </button>
   189→            <button
   190→              onClick={next}
   191→              className="absolute right-1 top-1/2 -translate-y-1/2 w-8 h-8 flex items-center justify-center bg-black/70 border-2 border-white text-white hover:bg-white hover:text-black transition-colors"
   192→              aria-label="Next image"
   193→            >
   194→              <ChevronRight className="w-4 h-4" />
   195→            </button>
   196→          </>
   197→        )}
   198→      </div>
   199→      {/* Dots */}
   200→      {images.length > 1 && (
   201→        <div className="flex justify-center gap-1 py-1.5 bg-black">
   202→          {images.map((_, i) => (
   203→            <button
   204→              key={i}
   205→              onClick={() => setIdx(i)}
   206→              className={cn(
   207→                "w-2 h-2 border border-white transition-colors",
   208→                i === idx ? "bg-white" : "bg-transparent",
   209→              )}
   210→              aria-label={`Go to image ${i + 1}`}
   211→            />
   212→          ))}
   213→        </div>
   214→      )}
   215→    </div>
   216→  );
   217→}

   219→function AchievementModal({
   220→  achievement,
   221→  memberColor,
   222→  onClose,
   223→}: {
   224→  achievement: Achievement | null;
   225→  memberColor: string;
   226→  onClose: () => void;
   227→}) {
   228→  const open = achievement !== null;
   229→  const panelRef = useRef<HTMLDivElement>(null);
   230→  useFocusTrap(panelRef, open);
   231→
   232→  useEffect(() => {
   233→    if (!open) return;
   234→    const onKey = (e: KeyboardEvent) => {
   235→      if (e.key === "Escape") {
   236→        document.body.classList.remove("modal-open");
   237→        onClose();
   238→      }
   239→    };
   240→    window.addEventListener("keydown", onKey);
   241→    return () => window.removeEventListener("keydown", onKey);
   242→  }, [open, onClose]);
   243→
   244→  // Hide navbar when modal is open
   245→  useEffect(() => {
   246→    if (open) {
   247→      document.body.classList.add("modal-open");
   248→    } else {
   249→      document.body.classList.remove("modal-open");
   250→    }
   251→    return () => { document.body.classList.remove("modal-open"); };
   252→  }, [open]);
   253→
   254→  return (
   255→    <AnimatePresence>
   256→      {open && achievement && (
   257→        <motion.div
   258→          className="fixed inset-0 z-[85] flex items-center justify-center p-4 md:p-8"
   259→          initial={{ opacity: 0 }}
   260→          animate={{ opacity: 1 }}
   261→          exit={{ opacity: 0 }}
   262→          transition={{ duration: 0.2 }}
   263→        >
   264→          <div className="absolute inset-0 bg-black/80 backdrop-blur-sm" onClick={onClose} />
   265→
   266→          {/* Relative wrapper */}
   267→          <div className="relative w-full max-w-md">
   268→            {/* X Close */}
   269→            <button
   270→              onClick={onClose}
   271→              className="absolute -top-4 -right-4 z-30 w-10 h-10 flex items-center justify-center bg-[#ff4d4d] text-white border-4 border-black dark:border-white shadow-[4px_4px_0_#000] dark:shadow-[4px_4px_0_#fff] hover:rotate-90 transition-transform no-color-transition"
   272→              aria-label="Close"
   273→            >
   274→              <X className="w-5 h-5" />
   275→            </button>
   276→
   277→          <motion.div
   278→            ref={panelRef}
   279→            role="dialog"
   280→            aria-modal="true"
   281→            aria-labelledby="ud-achievement-title"
   282→            tabIndex={-1}
   283→            className="relative w-full max-h-[90vh] overflow-y-auto border-8 border-black dark:border-white bg-white dark:bg-[#09090b] shadow-[8px_8px_0_#000] md:shadow-[16px_16px_0_#000] dark:md:shadow-[16px_16px_0_#d4ff00] outline-none"
   284→            initial={{ scale: 0.85, y: 30, rotate: -2 }}
   285→            animate={{ scale: 1, y: 0, rotate: 0 }}
   286→            exit={{ scale: 0.85, y: 30, rotate: -2 }}
   287→            transition={{ type: "spring", stiffness: 300, damping: 24 }}
   288→            onAnimationComplete={() => {
   289→              if (panelRef.current) panelRef.current.scrollTop = 0;
   290→            }}
   291→          >
   292→
   293→            {/* Colored header banner */}
   294→            <div className="p-6 border-b-8 border-black dark:border-white" style={{ backgroundColor: memberColor }}>
   295→              <div className="flex items-center justify-center mb-2">
   296→                <div className="w-16 h-16 flex items-center justify-center bg-black border-4 border-black">
   297→                  <Trophy className="w-8 h-8 text-white" />
   298→                </div>
   299→              </div>
   300→              <div className="font-mono-ud text-sm font-black tracking-[0.3em] uppercase text-black/60 text-center">
   301→                ▸ ACHIEVEMENT UNLOCKED
   302→              </div>
   303→            </div>
   304→
   305→            {/* Body */}
   306→            <div className="p-6">
   307→              <h3
   308→                id="ud-achievement-title"
   309→                className="font-bebas text-4xl md:text-5xl text-black dark:text-white leading-none mb-2 text-center"
   310→              >
   311→                {achievement.title}
   312→              </h3>
   313→              <div className="font-mono-ud text-xs font-black text-black/50 dark:text-white/50 tracking-widest mb-4 text-center">
   314→                ◆ {achievement.year} ◆
   315→              </div>
   316→              <div className="border-4 border-black dark:border-white bg-black dark:bg-white p-4 mb-4">
   317→                <p className="font-mono-ud text-sm text-white dark:text-black leading-relaxed text-center">
   318→                  {achievement.description}
   319→                </p>
   320→              </div>
   321→              {/* Achievement photo carousel (sertifikat, medali, foto kemenangan) */}
   322→              {achievement.images && achievement.images.length > 0 && (
   323→                <AchievementImageCarousel images={achievement.images} color={memberColor} />

   325→            </div>
   326→          </motion.div>
   327→          </div>
   328→        </motion.div>
   329→      )}
